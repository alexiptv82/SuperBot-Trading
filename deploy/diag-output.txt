=== hitv-proxy.conf ===
# HI-TV Prime — shared proxy directives
# Deploy target: /etc/nginx/snippets/hitv-proxy.conf
#
# The relay streams live TV and VOD through this proxy. Every directive
# below is load-bearing for playback — do not "tidy" them away.

proxy_pass         http://127.0.0.1:8001;
proxy_http_version 1.1;

# relay_service._compute_relay_base_url reads these two headers to build
# the relayBaseUrl it hands the Android client in the bootstrap
# response. Get them wrong and the app is handed an unreachable relay.
proxy_set_header   X-Forwarded-Proto $scheme;
proxy_set_header   X-Forwarded-Host  $host;

proxy_set_header   Host              $host;
proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
proxy_set_header   Connection        "";

# Streaming correctness. Buffering would stall live playback; the long
# timeouts match the relay's own httpx read=None (open-ended streams).
proxy_buffering         off;
proxy_request_buffering off;
proxy_read_timeout      3600s;
proxy_send_timeout      3600s;

# Never recompress .ts / .m3u8 payloads.
gzip off;

# Only POST /api/relay/bootstrap carries a body, and it is tiny.
client_max_body_size 1m;
