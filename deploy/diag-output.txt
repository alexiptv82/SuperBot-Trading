# HI-TV Prime — final nginx reverse proxy (HTTPS)
# Deploy target: /etc/nginx/sites-available/hitvprime
# Installed automatically by issue-cert.sh AFTER certificate issuance.

server {
    listen 80;
    listen [::]:80;
    server_name hitvprime.duckdns.org;
    access_log off;

    # Keep HTTP-01 reachable for future ACME renewals.
    location ^~ /.well-known/acme-challenge/ {
        root /var/www/acme;
        default_type "text/plain";
        try_files $uri =404;
    }

    location / {
        return 301 https://$host$request_uri;
    }
}

server {
    listen 443 ssl;
    listen [::]:443 ssl;
    http2 on;
    server_name hitvprime.duckdns.org;
    access_log off;

    ssl_certificate     /srv/hitv/tls/hitvprime.duckdns.org/fullchain.pem;
    ssl_certificate_key /srv/hitv/tls/hitvprime.duckdns.org/privkey.pem;

    # TLSv1.2 must remain enabled for legacy Android / Fire OS clients.
    ssl_protocols TLSv1.2 TLSv1.3;

    # RSA-capable suites retained for Android 5.1 / API 22 compatibility.
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;

    ssl_session_timeout 1d;
    ssl_session_cache shared:HiTVSSL:10m;
    ssl_session_tickets off;

    # No HSTS during migration/rollback window.

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/acme;
        default_type "text/plain";
        try_files $uri =404;
    }

    location / {
        include /etc/nginx/snippets/hitv-proxy.conf;
    }
}
